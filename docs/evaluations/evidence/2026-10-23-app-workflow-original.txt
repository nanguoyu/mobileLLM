function stripFragment(u) {
  const i = u.indexOf('#');
  if (i < 0) return u;
  return u.substring(0, i);
}

function hostOf(u) {
  const s = stripFragment(u);
  const i = s.indexOf('://');
  if (i < 0) return '';
  const start = i + 3;
  let end = s.length;
  for (let j = start; j < s.length; j++) {
    const c = s.charAt(j);
    if (c === '/' || c === '?' || c === '#') {
      end = j;
      break;
    }
  }
  return s.substring(start, end);
}

function sourceLabel(k) {
  if (k === 'page-excerpt') return '页面摘录';
  if (k === 'search-summary') return '搜索摘要';
  return '未观察';
}

    s += 'URL: ' + priorCitationReceipts[i].url + ' 类型: ' + priorCitationReceipts[i].kind + '\n';
  }
  s += '备注: https://www.aalto.fi/en/people/bo-zhao 的页面摘录仅为 Cookie 提示，不能作为研究正文。\n';
  return s;
}

function kindOfFor(url) {
  const key = stripFragment(url);
  const p = priorMap[key] || '';
  const c = currentMap[key] || '';
  if (p === 'page-excerpt' || c === 'page-excerpt') return 'page-excerpt';
  if (p === 'search-summary' || c === 'search-summary') return 'search-summary';
  return '';
}

function sourceNote(r) {
  const off = kindOfFor(r.officialSource);
  const work = kindOfFor(r.workSource);
  return '；来源标注：官方源=' + sourceLabel(off) + '，成果源=' + sourceLabel(work);
}

const teamCount = rows.length;
const countryCount = unique(rows.map(function(r) { return r.countryCode; })).length;

const hostList = [];
for (let i = 0; i < rows.length; i++) {
  const row = rows[i];
  if (row.officialSource.indexOf('https://') === 0) {
    const oh = hostOf(row.officialSource);
    if (oh !== '' && hostList.indexOf(oh) < 0) hostList.push(oh);
  }
  if (row.workSource.indexOf('https://') === 0) {
    const wh = hostOf(row.workSource);
    if (wh !== '' && hostList.indexOf(wh) < 0) hostList.push(wh);
  }
}
const hostCount = hostList.length;
const statsLine = '覆盖统计：团队 ' + teamCount + ' 个，国家代码 ' + countryCount + ' 个，HTTPS 引用主机 ' + hostCount + ' 个。';

function limitationsText(list) {
  let s = '';
  for (let i = 0; i < list.length; i++) {
    s += '- ' + list[i] + '\n';
  }
  return s;
}
